async function persiapkanMulaiUjian() { const ruang = document.getElementById("selectRuang").value; const kelas = document.getElementById("selectKelas").value; const selectNama = document.getElementById("selectNama"); const nama = selectNama.value; const nis = selectNama.selectedOptions[0]?.dataset.nis || ""; const selectUjian = document.getElementById("selectUjian"); const ujianIndex = selectUjian.value; const btnSubmit = document.getElementById("btnSubmitMulai"); if (!ruang || !kelas || !nama) { alert("Harap lengkapi identitas diri terlebih dahulu!"); return; } if (ujianIndex === "" || !currentAvailableUjian[ujianIndex]) { alert("Silakan pilih mata pelajaran / ujian yang akan dikerjakan!"); return; } const ujianTerpilih = currentAvailableUjian[ujianIndex]; if (!ujianTerpilih.linkForm) { alert("Tautan soal Google Form belum disetel di tab UJIAN. Hubungi pengawas."); return; } // Ubah tombol ke status memuat btnSubmit.disabled = true; const originalBtnText = btnSubmit.innerHTML; btnSubmit.innerHTML = `Memeriksa sesi...`; let sessionRecovery = { isResumed: false, totalPelanggaran: 0, isLocked: false }; try { // 1. Verifikasi Status Siswa ke Server Google Sheets const checkUrl = `${SCRIPT_URL}?action=checkSession&nama=${encodeURIComponent(nama)}&mapel=${encodeURIComponent(ujianTerpilih.mapel)}`; const res = await fetch(checkUrl); const data = await res.json(); if (data.status === "success" && data.session) { const s = data.session; // Tolak login ulang jika siswa sudah menyelesaikan ujian secara sah if (s.isCompleted) { alert("Anda sudah menyelesaikan ujian untuk mata pelajaran ini dan jawaban telah dikirim. Akses tidak dapat diulang."); btnSubmit.disabled = false; btnSubmit.innerHTML = originalBtnText; return; } // Jika siswa pernah masuk sebelumnya dan belum selesai (tutup tab) if (s.hasSession) { sessionRecovery.isResumed = true; sessionRecovery.totalPelanggaran = s.totalPelanggaran; sessionRecovery.isLocked = s.isLocked; } } } catch (e) { console.warn("Koneksi verifikasi server terhambat, memeriksa cache lokal:", e); // Fallback ke localStorage jika koneksi terganggu const localDone = localStorage.getItem(`exam_done_${nama}_${ujianTerpilih.mapel}`); if (localDone === "true") { alert("Ujian ini sudah diselesaikan sebelumnya."); btnSubmit.disabled = false; btnSubmit.innerHTML = originalBtnText; return; } const localV = localStorage.getItem(`exam_v_${nama}_${ujianTerpilih.mapel}`); if (localV !== null) { sessionRecovery.isResumed = true; sessionRecovery.totalPelanggaran = parseInt(localV, 10); sessionRecovery.isLocked = localStorage.getItem(`exam_lock_${nama}_${ujianTerpilih.mapel}`) === "true"; } } // Simpan parameter sesi ke sessionStorage sessionStorage.setItem("exam_ruang", ruang); sessionStorage.setItem("exam_kelas", kelas); sessionStorage.setItem("exam_nama", nama); sessionStorage.setItem("exam_nis", nis); sessionStorage.setItem("exam_mapel", ujianTerpilih.mapel); sessionStorage.setItem("exam_link_form", ujianTerpilih.linkForm); sessionStorage.setItem("exam_token", configData.tokenUjian); sessionStorage.setItem("exam_token_reset", configData.tokenResetPengawas); sessionStorage.setItem("exam_max_violation", configData.batasPelanggaran); // Pasang hasil pemulihan sesi (bukan angka nol buta) sessionStorage.setItem("exam_pelanggaran", sessionRecovery.totalPelanggaran.toString()); sessionStorage.setItem("exam_is_locked", sessionRecovery.isLocked ? "true" : "false"); sessionStorage.setItem("exam_is_resumed", sessionRecovery.isResumed ? "true" : "false"); sessionStorage.setItem("exam_script_url", SCRIPT_URL); window.location.href = "exam.html"; }
